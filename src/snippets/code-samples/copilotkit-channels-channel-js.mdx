```ts icon="server" title="channel.ts"
import { createChannel, Message, Section } from "@copilotkit/channels";
import { slack } from "@copilotkit/channels/slack";

import { makeAgent } from "./agent"; // your LangGraph AG-UI agent

const channel = createChannel({
  // Point the channel at your LangChain agent (an AbstractAgent, or a per-thread factory).
  agent: makeAgent,
  adapters: [
    slack({
      botToken: process.env.SLACK_BOT_TOKEN!, // xoxb-…
      appToken: process.env.SLACK_APP_TOKEN!, // xapp-… (Socket Mode)
    }),
  ],
});

// A mention starts a turn; runAgent streams the reply back to Slack.
channel.onMention(async ({ thread }) => {
  await thread.runAgent();
});

await channel.start();
```
