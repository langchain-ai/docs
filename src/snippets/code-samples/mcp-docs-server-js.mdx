```typescript
import { createAgent } from "langchain";
import { MCPAdapter } from "@langchain/mcp-adapters";

const adapter = new MCPAdapter("https://docs.langchain.com/mcp");  // # [!code highlight]
const tools = await adapter.listTools();
const agent = createAgent({ model: "claude-sonnet-5", tools: [tools] })

await agent.invoke({
  messages: [{
    role: "user",
    content: "How do I add short-term memory to a LangChain agent?"
  }]
})
```
