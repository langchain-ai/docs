import asyncio

from langsmith import Client


async def main():
    client = Client()
    queue_id = "<queue-id>"
    runs = list(client.list_runs(project_name="default", limit=5))
    await client.annotation_queues.items.create(
        queue_id,
        items=[
            {
                "item_type": "RUN",
                "run_id": str(run.id),
                "project_id": str(run.session_id),
                "start_time": run.start_time.isoformat(),
            }
            for run in runs
        ],
    )


asyncio.run(main())
