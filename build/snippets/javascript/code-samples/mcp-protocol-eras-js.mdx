```typescript
import { MCPAdapter } from "@langchain/mcp-adapters";

const adapter = new MCPAdapter({
  servers: {
    current: {
      url: "https://example.com/mcp",
      mode: "modern",
    },
    legacy: {
      command: "node",
      args: ["./legacy-server.js"],
      mode: "legacy",
    },
  },
});

try {
  const tools = await adapter.listTools();
  console.log(tools);
} finally {
  await adapter.close();
}
```
