      model,
    };

      model,
      tools: [],

        createFilesystemMiddleware({ backend }),
        createSummarizationMiddleware({ model, backend }),
        createSkillsMiddleware({ backend, sources: ["/skills/"] }),
        todoListMiddleware(),

          defaultModel: model,
          defaultTools: [],
          subagents: [visualizer],
        }),
      ],
    });
