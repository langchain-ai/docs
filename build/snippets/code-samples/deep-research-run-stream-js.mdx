async function main() {
  const stream = await agent.streamEvents(
    {
      messages: [
        {
          role: "user",
          content: "Compare Python vs JavaScript for web development",
        },
      ],
    },
    { version: "v3" },
  );

  for await (const message of stream.messages) {
    process.stdout.write(await message.text);
  }

  await stream.output;
}

main().catch((err) => {
  console.error(err);
  process.exitCode = 1;
});
