```kotlin Kotlin
import com.langchain.smith.client.okhttp.LangsmithOkHttpClient
import com.langchain.smith.evaluation.EvaluationResult
import com.langchain.smith.evaluation.EvaluateParams
import com.langchain.smith.evaluation.evaluate
import com.langchain.smith.evaluation.runEvaluator

fun main() {
    val client = LangsmithOkHttpClient.fromEnv()

    // Your app: turn each dataset row's inputs into an output.
    val myApp: (Map<String, Any?>) -> Map<String, Any?> = { inputs ->
        mapOf("answer" to inputs["question"])
    }

    // Your scorer: compare the app's answer to the expected answer.
    val correctness = runEvaluator { outputs, reference ->
        val score =
            if (outputs["answer"] == reference["output"]) 1.0 else 0.0
        EvaluationResult(key = "correct", score = score)
    }

    val params =
        EvaluateParams.builder()
            .data("my-dataset") // dataset name or id in LangSmith
            .experimentPrefix("my-first-test")
            .addEvaluator(correctness)
            .build()

    val results = evaluate(client, myApp, params)

    println("Experiment: ${results.experimentName}")
    println("Rows scored: ${results.rows.size}")
}
```
