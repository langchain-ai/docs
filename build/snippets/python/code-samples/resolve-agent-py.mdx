```python
import asyncio

import langsmith as ls

client = ls.Client()
agent = ls.AgentAddress("checkout", "production")


async def main() -> None:
    project = await client.sessions.resolve(**agent.to_api_address())

    async for run in client.runs.query_v2(project_ids=[project.session_id]):
        print(run.id)


asyncio.run(main())
```
